# Install with:  pip install -r requirements.txt
# Optional: ffmpeg on PATH enables M4B audiobook export.

requests        # makes HTTP calls to the DeepInfra API
python-dotenv   # loads the API key from the .env file
num2words       # turns numbers into words ("1200" -> "one thousand two hundred") for normalize.py
fastapi         # the web framework: routes, request validation
uvicorn         # the server that runs the FastAPI app
pypdf           # reads text out of PDF files, for extract.py
python-multipart # lets FastAPI receive file uploads
ebooklib        # parses EPUB structure (chapters + reading order), for extract.py
beautifulsoup4  # strips HTML tags from EPUB chapters down to text
